cmake_minimum_required(VERSION 3.19)
project(valen_cpp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ORT_LIBRARY "" CACHE FILEPATH "ONNX Runtime 1.31 shared library")
set(VALEN_COMPONENT_LIBRARY "${CMAKE_CURRENT_SOURCE_DIR}/native/target/release/libvalen_text_components.a" CACHE FILEPATH "Built Rust tokenizer/template components")
if(NOT EXISTS "${ORT_LIBRARY}" OR NOT EXISTS "${VALEN_COMPONENT_LIBRARY}")
  message(FATAL_ERROR "Set ORT_LIBRARY and build the native components first")
endif()
enable_testing()
add_executable(valen-cpp main.cpp)
add_executable(valen-cpp-tests tests/decision15_test.cpp)
add_test(NAME decision15 COMMAND valen-cpp-tests)
if(APPLE)
 file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
 file(CREATE_LINK "${ORT_LIBRARY}" "${CMAKE_BINARY_DIR}/lib/libonnxruntime.1.dylib" SYMBOLIC)
else()
 get_filename_component(ORT_LIBRARY_DIR "${ORT_LIBRARY}" DIRECTORY)
endif()
foreach(target valen-cpp valen-cpp-tests)
 target_include_directories(${target} PRIVATE vendor/onnxruntime vendor)
 target_link_libraries(${target} PRIVATE "${VALEN_COMPONENT_LIBRARY}" "${ORT_LIBRARY}")
 if(APPLE)
  target_link_libraries(${target} PRIVATE "-framework Security" "-framework CoreFoundation")
  set_target_properties(${target} PROPERTIES BUILD_RPATH "${CMAKE_BINARY_DIR}/lib")
 else()
  target_link_libraries(${target} PRIVATE dl pthread m)
  set_target_properties(${target} PROPERTIES BUILD_RPATH "${ORT_LIBRARY_DIR}")
 endif()
endforeach()
